## 구현한 것

- 배열을 선언하고 스폰/디스폰 과정을 통해 플레이어 주위에 음표(노트) 오브젝트들을 생성해 이펙트 표현
- 오선지 다리를 지날 때 음표들을 파티클 시스템으로 떠오르게 구현
- 파티클 시스템에 콜라이더를 붙여 플레이어가 지나갈 때만 파티클이 나오도록 구현(Trigger Enter/Exit)

## 배운 것(개념/정리)

- 오브젝트 풀링 구조에서는 OnEnable을 “재활성화될 때 초기화 지점”으로 쓰면 타이머/랜덤값/머티리얼 세팅을 깔끔하게 관리할 수 있음
- 머티리얼 색상 적용은 셰이더에 따라 \_BaseColor / \_Color 등 프로퍼티가 달라질 수 있어 방어적으로 처리하는 게 안전함

## 이슈/버그

- 내용: 오브젝트 풀링 방식으로 노트가 주위에 떠다니는 효과를 만들 때 만들어진 노트들이 투명해서 잘 보이지 않는 문제 발생
- 원인/가설: 색 배열(Color\[\]) 안의 알파(A) 값이 0(또는 낮은 값)으로 들어가 있었을 가능성이 큼
- 해결/시도: 시작 색상의 알파를 1로 강제 고정하고, 페이드아웃 구간에서만 0으로 선형 보간되도록 로직 수정(주석 추가 및 구조 정리)
- 다음 액션: 머티리얼 컬러 프로퍼티(\_BaseColor / \_Color) 대응을 유지하고, 인스펙터에 넣는 색상의 알파값을 사전 점검하는 체크리스트화

- NoteVisuallizer (기존 버전: 투명도 문제 발생)

```cs
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class NoteVisuallizer : MonoBehaviour
{
    [Header("수명")]
    [SerializeField] private float lifeTime = 2f;

    [Header("이동")]
    [SerializeField] private float floatSpeedMin = 0.3f;
    [SerializeField] private float floatSpeedMax = 0.8f;

    [Header("크기")]
    [SerializeField] private float scaleMin = 0.7f;
    [SerializeField] private float scaleMax = 1.3f;

    [Header("회전")]
    [SerializeField] private float rotSpeedMin = -20f;
    [SerializeField] private float rotSpeedMax = 20f;

    [Header("랜덤 색상 후보")]
    [SerializeField] private Color[] colors;

    private float timer;
    private float floatSpeed;
    private float rotSpeed;

    private MeshRenderer meshRenderer;
    private Material Mat;
    private Color startColor;
    private Vector3 startScale;

    private void Start()
    {
        meshRenderer = GetComponent<MeshRenderer>();
    }
    private void OnEnable()
    {

        if (meshRenderer == null)
        {
            meshRenderer = GetComponent<MeshRenderer>();
        }
        if (meshRenderer != null)
        {
            Mat = meshRenderer.material;

        }

        timer = 0f;
        // 랜덤 크기 만들기
        float randomScale = Random.Range(scaleMin , scaleMax);
        startScale = Vector3.one * randomScale;
        transform.localScale = startScale;
        // 랜덤 속도
        floatSpeed = Random.Range(floatSpeedMin, floatSpeedMax);

        // 랜덤 회전
        rotSpeed = Random.Range(rotSpeedMin, rotSpeedMax);

        // 랜덤 색상
        if (Mat != null)
        {
            if (colors != null && colors.Length > 0)
            {
                startColor = colors[Random.Range(0, colors.Length)];
            }
            else
            {
                startColor = new Color(Random.value, Random.value, Random.value);
            }
            Mat.color = startColor;
        }
    }
    private void Update()
    {
        timer += Time.deltaTime;

        // 위로 이동
        transform.Translate(Vector3.up * floatSpeed * Time.deltaTime, Space.World);
        // 회전
        transform.Rotate(Vector3.forward * rotSpeed * Time.deltaTime, Space.World);
        // 투명도 감소
        if (Mat != null)
        {
            float t = Mathf.Clamp01(timer / lifeTime);
            float alpha = Mathf.Lerp(startColor.a, 0f, t);
            Color fadeColor = new Color(startColor.r, startColor.g, startColor.b, alpha);
            Mat.color = fadeColor;
        }

        // 수명 종료 시 비활성화
        if (timer >= lifeTime)
        {
            gameObject.SetActive(false);
        }
    }
}
```

- NoteVisuallizer (수정 후: 알파 1 고정 + 페이드아웃 + 주석 보강)

```cs
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

// 음표 오브젝트가
// 1. 랜덤한 색을 갖고
// 2. 랜덤한 크기를 갖고
// 3. 위로 떠오르면서 회전하다가
// 4. 서서히 투명해진 뒤
// 5. 일정 시간이 지나면 비활성화되는 역할을 하는 스크립트
public class NoteVisuallizer : MonoBehaviour
{
    [Header("수명")]
    [SerializeField] private float lifeTime = 2f;

    [Header("이동")]
    [SerializeField] private float floatSpeedMin = 0.3f;
    [SerializeField] private float floatSpeedMax = 0.8f;

    [Header("크기")]
    [SerializeField] private float scaleMin = 0.7f;
    [SerializeField] private float scaleMax = 1.3f;

    [Header("회전")]
    [SerializeField] private float rotSpeedMin = -20f;
    [SerializeField] private float rotSpeedMax = 20f;

    [Header("랜덤 색상 후보")]
    [SerializeField] private Color[] colors;

    private float timer;
    private float floatSpeed;
    private float rotSpeed;

    private MeshRenderer meshRenderer;
    private Material noteMat;
    private Color startColor;

    private void Awake()
    {
        meshRenderer = GetComponent<MeshRenderer>();
    }

    private void OnEnable()
    {
        if (meshRenderer == null)
            meshRenderer = GetComponent<MeshRenderer>();

        if (meshRenderer != null)
            noteMat = meshRenderer.material;

        timer = 0f;

        float randomScale = Random.Range(scaleMin, scaleMax);
        transform.localScale = Vector3.one * randomScale;

        floatSpeed = Random.Range(floatSpeedMin, floatSpeedMax);
        rotSpeed = Random.Range(rotSpeedMin, rotSpeedMax);

        SetRandomColor();
    }

    private void Update()
    {
        timer += Time.deltaTime;

        transform.Translate(Vector3.up * floatSpeed * Time.deltaTime, Space.World);
        transform.Rotate(Vector3.forward * rotSpeed * Time.deltaTime, Space.World);

        FadeOut();

        if (timer >= lifeTime)
            gameObject.SetActive(false);
    }

    private void SetRandomColor()
    {
        if (noteMat == null) return;

        if (colors != null && colors.Length > 0)
            startColor = colors[Random.Range(0, colors.Length)];
        else
            startColor = new Color(Random.value, Random.value, Random.value, 1f);

        startColor.a = 1f;
        ApplyColor(startColor);
    }

    private void FadeOut()
    {
        if (noteMat == null) return;

        float t = Mathf.Clamp01(timer / lifeTime);
        float alpha = Mathf.Lerp(1f, 0f, t);

        Color fadeColor = new Color(startColor.r, startColor.g, startColor.b, alpha);
        ApplyColor(fadeColor);
    }

    private void ApplyColor(Color color)
    {
        if (noteMat == null) return;

        if (noteMat.HasProperty("_BaseColor"))
            noteMat.SetColor("_BaseColor", color);
        else if (noteMat.HasProperty("_Color"))
            noteMat.SetColor("_Color", color);
    }
}
```

- PopNotes (플레이어 트리거로 파티클 재생/정지)

```cs
using UnityEngine;

public class PopNotes : MonoBehaviour
{
    [SerializeField] private ParticleSystem noteParticle;
    private BoxCollider boxCol;

    private void Start()
    {
        if (noteParticle == null)
            noteParticle = GetComponentInChildren<ParticleSystem>();

        if (noteParticle != null)
            noteParticle.Stop(true, ParticleSystemStopBehavior.StopEmittingAndClear);

        boxCol = GetComponent<BoxCollider>();
    }

    private void OnTriggerEnter(Collider col)
    {
        if (col.CompareTag("Player"))
        {
            if (noteParticle != null)
                noteParticle.Play();

            Debug.Log("플레이어 들어옴!");
        }
    }

    private void OnTriggerExit(Collider col)
    {
        if (col.CompareTag("Player"))
        {
            if (noteParticle != null)
                noteParticle.Stop(true, ParticleSystemStopBehavior.StopEmitting);

            Debug.Log("플레이어 나감!");
        }
    }
}
```

<summary>회고(선택)</summary>

- 잘한 점: 현상 → 원인 가설 → 해결 → 다음 액션 순서로 정리하면서 빠르게 문제를 좁히고 수정했다. 주석을 보강해서 다음에 봐도 이해 가능하게 만들어뒀다.
- 아쉬운 점: 인스펙터에 넣는 색상 데이터(알파값) 같은 세팅 실수를 초기에 체크하지 못했다.
- 내일 개선: 기능 구현 전에 “데이터/머티리얼/셰이더 세팅 점검”을 먼저 하고, 재사용 가능한 이펙트/풀링 스크립트는 템플릿화한다.
