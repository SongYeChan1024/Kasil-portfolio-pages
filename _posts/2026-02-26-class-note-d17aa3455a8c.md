<summary>인벤토리 드래그·드롭 연결</summary>

어제 만들었던 Drag 에 아이템 리스트의 트랜스폼과 캔버스 그룹을 설정하여 블락 레이캐스트를 설정해서 다른 이벤트를 받지 않도록 하고 부모를 바꿔서 슬롯으로 이동 하도록 구현

```cs
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.EventSystems;
//드래그 기능 구현
//1.드래그할 아이템의 트랜스폼
//2.Grid Layout Group의 영향을 받지 않도록 하기 위해 드래그할 아이템의 부모를 인벤토리로 바꾸기 위한 인벤토리의 트랜스폼
//3.드래그 중인 아이템을 저장하는 static 변수
//4.드래그를 하다가 드롭이 안될 때 아이템의 부모를 아이템 리스트로 바꾸기 위한 아이템 리스트의 트랜스폼
//5.드래그 중인 아이템의 CanvasGroup 컴포넌트
public class Drag : MonoBehaviour, IDragHandler, IBeginDragHandler, IEndDragHandler
//드래그 이벤트 인터페이스 구현
{
    private Transform itemTr;//드래그 할 아이템의 트랜스폼

    public Transform inventoryTr; //인벤토리의 트랜스폼

    public Transform itemListTr; //아이템 리스트의 트랜스폼

    public static GameObject draggingitem = null; //드래그 중인 아이템을 저장하는 static 변수

    private CanvasGroup canvasGroup; //드래그 중인 아이템의 CanvasGroup 컴포넌트
    void Start()
    {
        itemTr = GetComponent<Transform>(); //드래그 할 아이템의 트랜스폼 할당
        //inventoryTr = GameObject.Find("Inventory").GetComponent<Transform>();
        //인벤토리의 트랜스폼 할당 근데 파인드로 할당하면 메모리를 많이 잡아먹으니까 직접 할당
        //특히 void Update()에서는 절대 할당 하면 안됨 왜냐하면 업데이트는 매 프레임마다 실행되기 때문에 메모리를 미친듯이 잡아먹음
        canvasGroup = GetComponent<CanvasGroup>(); 
        //드래그 중인 아이템의 CanvasGroup 컴포넌트 할당
    }
    public void OnDrag(PointerEventData eventData) //드래그 이벤트
    {
        itemTr.position = Input.mousePosition; 
        //드래그를 하면 아이템이 마우스의 위치에 따라 이동
    }

    public void OnBeginDrag(PointerEventData eventData) //드래그를 시작할 때 부모인 아이템 리스트의 Grid Layout Group의 영향을 받지 않도록 하기 
    {
        this.transform.SetParent(inventoryTr);
        //드래그를 시작할 때 아이템의 부모를 인벤토리로 설정하여 Grid Layout Group의 영향을 받지 않도록 함
        draggingitem = this.gameObject; 
        //드래그 중인 아이템을 저장하는 static 변수에 현재 드래그 중인 아이템 할당
        canvasGroup.blocksRaycasts = false;
        //드래그가 시작 되면 아이템이 다른 UI 요소의 Event를 받지 않도록 설정

    }

    public void OnEndDrag(PointerEventData eventData) //드래그를 끝낼 때 아이템이 인벤토리의 영역 밖으로 벗어나지 않도록 하고 Grid Layout Group의 영향을 받도록 하기
    {
        draggingitem = null; //드래그 중인 아이템을 저장하는 static 변수 초기화

        canvasGroup.blocksRaycasts = true;//드래그가 끝나면 아이템이 다른 UI 요소의 Event를 받을 수 있도록 설정

        if (itemTr.parent == inventoryTr) //드래그를 끝낼 때 아이템이 인벤토리의 영역 밖으로 벗어나지 않도록 하기 위해 아이템의 부모가 인벤토리인 경우에만 드래그를 끝냄
        {
            this.transform.SetParent(itemListTr);
            //드래그를 끝낼 때 아이템의 부모를 아이템 리스트로 설정하여 Grid Layout Group의 영향을 받도록 함
        }
    }
}
```

<summary>2.인벤토리 버튼을 만들고 인벤토리를 끄고 킬 수 있게 구현</summary>

<details markdown="1">
<summary>게임매니저에 인벤토리의 캔버스 그룹을 넣어서 알파 값을 0이면 꺼지고 1이면 켜지게 구현</summary>

```txt
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using static Codice.Client.Common.Connection.AskCredentialsToUser;
//적이 태어나게 만들어 보자, 나중에 오브젝트 풀링을 할때는 오브젝트 풀링매니저를 만들어서 옮길 예정
//순서 1.태어날 위치(SpawnPoint)
//2.적 프리팹 오브젝트
//3.적이 태어나는 시간 간격(SpawnTime)
//4.적이 태어나는 최대 수(MaxEnemyCount)
//싱글턴 디자인 패턴으로 만들어 보기
//5.게임 일시정지 기능 추가
public class GameManager : MonoBehaviour
{
    public static GameManager instance = null; //싱글턴 인스턴스 변수
    [Header("적이 태어날 위치")]
    public Transform[] points;
    public bool isGameOver = false;
    
    [Header("적 프리팹 오브젝트")]
    public GameObject[] enemyPrifab;
    
    [Header("적이 태어나는 시간 간격")]
    private float CreateTime = 3f;
    private WaitForSeconds createWs;

    [Header("적이 태어나는 최대 수")]
    private int maxEnemy = 10;

    private readonly string[] enemyTag = { "ENEMY", "SWAT" };

    [Header("게임 일시정지")]
    private bool isPaused;

    public CanvasGroup inventoryCG; //인벤토리의 CanvasGroup 컴포넌트

    private void Awake()
    {
        if (instance == null) //인스턴스가 아직 생성되지 않은 경우
        {
            instance = this; //현재 객체를 인스턴스로 설정
        }
        else if (instance != this) //인스턴스가 이미 존재하고 현재 객체가 인스턴스가 아닌 경우
        {
            Destroy(gameObject); //현재 객체를 파괴하여 중복된 인스턴스 방지
            //다른 씬에 갔다가 원래 씬으로 돌아오면 GameManager가 또 생기는데, 그걸 방지하기 위해서 만든 코드
        }
        DontDestroyOnLoad(this.gameObject); //씬이 변경되어도 이 객체가 파괴되지 않도록 설정
    }
    void Start()
    {
        points = GameObject.Find("SpawnPoints").GetComponentsInChildren<Transform>();
        createWs = new WaitForSeconds(CreateTime);
        if (points.Length > 0 && enemyPrifab.Length > 0)
        {
            StartCoroutine(CreateEnemy());
        }
    }
    IEnumerator CreateEnemy()
    {        
        while (!isGameOver)
        {
            int enemyCount = 0; //현재 씬에 존재하는 적의 수를 세는 변수
            for (int i = 0; i < enemyTag.Length; i++) //적 태그 배열을 순회하면서
                enemyCount += GameObject.FindGameObjectsWithTag(enemyTag[i]).Length; //각 태그에 해당하는 게임 오브젝트의 수를 enemyCount에 더함
            if (enemyCount < maxEnemy) 
            {
                yield return createWs; //3초후에 태어나게
                int idx = Random.Range(1, points.Length); //0번은 SpawnPoints이므로 1부터 시작 //불규칙하게 태어나게
                Instantiate(enemyPrifab[Random.Range(0, enemyPrifab.Length)], points[idx].position, points[idx].rotation);
            }
            else
            {
                yield return null; //한 프레임 대기
            }
           
        }       
    }
    public void OnPauseClick()
    {
        isPaused = !isPaused; //토글 방식으로 일시정지 상태 변경
        Time.timeScale = (isPaused ? 0.0f : 1.0f);
        //Time.timeScale을 0으로 설정하면 게임이 일시정지되고, 1로 설정하면 게임이 정상적으로 진행

        // 플레이어가 전혀 움직이지 못하게 플레이어의 스크립트들을 비활성화
        var player = GameObject.FindGameObjectWithTag("Player");
        //GameObject.FindGameObjectWithTag("PLAYER") 메서드를 사용하여 "PLAYER" 태그가 지정된 게임 오브젝트를 찾아 player 변수에 할당
        MonoBehaviour[] scripts = player.GetComponents<MonoBehaviour>();
        //GetComponents<MonoBehaviour>() 메서드를 사용하여 플레이어 게임 오브젝트에 연결된 모든 MonoBehaviour 스크립트를 배열로 가져옴
      foreach (MonoBehaviour script in scripts)
        //foreach 루프를 사용하여 배열에 있는 각 스크립트에 대해 반복하면서,
        //script.enabled 속성을 isPaused의 반대값으로 설정하여 일시정지 상태에 따라 스크립트를 활성화하거나 비활성화
        {
            script.enabled = !isPaused; //일시정지 상태에 따라 스크립트 활성화/비활성화
        }
      var canvasGroup = GameObject.Find("Panel-Weapon").GetComponent<CanvasGroup>();
        canvasGroup.blocksRaycasts = !isPaused; //일시정지 상태에 따라 UI의 상호작용 가능 여부 설정
    }
    //인벤토리를 활성화 또는 비활성화하는 메서드
    public void OnInventoryOpen(bool isOpen)
    {
        inventoryCG.alpha = isOpen ? 1 : 0; //인벤토리가 열릴 때는 투명도를 1로, 닫힐 때는 0으로 설정
        inventoryCG.interactable = isOpen; //인벤토리가 열릴 때는 상호작용 가능, 닫힐 때는 불가능
        inventoryCG.blocksRaycasts = isOpen; //인벤토리가 열릴 때는 UI 요소가 레이캐스트를 차단하여 상호작용할 수 있도록 설정, 닫힐 때는 차단하지 않도록 설정
    }
}
```

<summary>캔버스 그룹에 대한 설명</summary>

<details markdown="1">
<summary><b>1.Alpha (알파)</b></summary>

<b>설명: 그룹 전체의 투명도를 조절합니다.</b>

<b>범위: 0.0(완전 투명) \~ 1.0(완전 불투명)</b>

<b>특징: 하위 자식들의 Image나 Text 컬러를 일일이 바꿀 필요 없이, 이 값만 조절하면 전체가 부드럽게 나타나거나 사라지게 할 수 있습니다. UI 페이드 인/아웃 연출을 할 때 필수입니다.</b>

</details>

<details markdown="1">
<summary><b>2.Interactable (상호작용 가능 여부)설명: 이 그룹에 포함된 UI 요소들이 클릭이나 입력에 반응할지 결정합니다.</b></summary>

Interactable (상호작용 가능 여부)설명: 이 그룹에 포함된 UI 요소들이 클릭이나 입력에 반응할지 결정합니다.

<b>체크 해제 시: 버튼 등이 시각적으로는 보이지만 클릭은 되지 않는 ‘비활성화’ 상태가 됩니다.</b>

(단, 시각적으로 회색 처리가 자동으로 되지는 않으니 코드로 알파값을 같이 조절하는 경우가 많습니다.)

</details>

<details markdown="1">
<summary><b>3.Blocks Raycasts (레이캐스트 차단)설명: UI가 마우스 클릭이나 터치 이벤트(Raycast)를 받을지 설정합니다.</b></summary>

Blocks Raycasts (레이캐스트 차단)설명: UI가 마우스 클릭이나 터치 이벤트(Raycast)를 받을지 설정합니다.

<b>체크 해제 시: 마우스 클릭이 이 UI를 통과하여 뒤에 있는 오브젝트나 다른 UI에 전달됩니다.</b>

<b>꿀팁: Alpha를 0으로 만들어 UI를 숨겼더라도, Blocks Raycasts가 켜져 있으면 눈에 안 보이는</b>

UI가 클릭을 막아버릴 수 있습니다. 숨길 때는 이 옵션도 같이 끄는 게 좋습니다.

</details>

<details markdown="1">
<summary><b>4.Ignore Parent Groups (부모 그룹 무시)설명: 부모 오브젝트에 또 다른 Canvas Group이 있을 때, 그 부모의 설정을 따를지 말지 결정합니다.</b></summary>

Ignore Parent Groups (부모 그룹 무시)설명: 부모 오브젝트에 또 다른 Canvas Group이 있을 때, 그 부모의 설정을 따를지 말지 결정합니다.

<b>체크 시: 부모 Canvas Group의 알파값이나 상호작용 설정이 바뀌어도 현재 그룹은 영향을 받지 않고 독자적으로 움직입니다</b>

</details>

<details markdown="1">
<summary>3.오브젝트 풀링</summary>

- 오브젝트 풀링을 하는 이유

모바일 플랫폼에서 게임 오브젝트나 프리팹을 동적으로 생성했다가 <br />Destroy 하는 방식은 물리적인 부하가 걸릴 수 밖에 없다.<br />끊임 없이 Instantiate() → Destroy : 물리적 부하 많이 걸림

<b>그래서 주기적으로 반복적으로 생성하는 객체는 씬이 처음 로드 될 때</b>

<b>오브젝트를 끈 상태로 모두 생성했다가 필요할 때 하나씩 오브젝트를</b>

<b>활성화하고 필요에 의해서 오브젝트를 비활성화하는 방식</b>

<details markdown="1">
<summary>총알 오브젝트 풀링(Bullet Object Pooling)</summary>

<details markdown="1">
<summary>이전에 만들었던 에너미 데미지와 스왓 데미지 에서 오류가 나는 부분을 해결 하기위해 디스트로이를 주석 처리 하고 풀링으로 구현</summary>

```txt
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class EnemyDamage : MonoBehaviour
{
    private const string bulletTag = "BULLET"; //총알 태그 상수
    private float hp = 100f; //에너미의 체력 변수
    [SerializeField] private GameObject bloodEffect; //피격 효과 프리팹 변수

    void Start()
    {
        bloodEffect = Resources.Load("Effects/BulletImpactFleshBigEffect") as GameObject;
        //Resources.Load는 프로젝트의 Resources 폴더에서 리소스를 로드하는 메서드.
        //"Effects/BulletImpactFleshBigEffect"이라는 이름의 GameObject를 로드하여 bloodEffect 변수에 할당.
    }
    private void OnCollisionEnter(Collision col)
    {
        if (col.collider.CompareTag(bulletTag)) //충돌한 객체가 총알 태그를 가지고 있는지 확인
        {
            //Destroy(col.gameObject); //충돌한 총알 객체를 파괴
            col.gameObject.SetActive(false); //충돌한 총알 오브젝트 비활성화
            hp -= col.gameObject.GetComponent<BulletCTRL>().damage; //BulletCTRL 스크립트에서 damage 값을 가져와 hp에서 차감
            if (hp <= 0.0f) //체력이 0 이하로 떨어지면
            {
                GetComponent<EnemyAI>().Die();//EnemyAI 스크립트의 Die() 메서드를 호출하여 에너미를 죽이는 처리
            }
            Vector3 pos = col.contacts[0].point; //충돌 지점의 위치를 pos 변수에 저장
            Vector3 _normal = col.contacts[0].normal; //충돌 지점의 법선 벡터를 normal 변수에 저장
            Quaternion rot = Quaternion.FromToRotation(-Vector3.forward, _normal);
            //총알이 충돌한 지점의 법선 벡터를 기준으로 회전값을 계산하여 rot 변수에 저장

            var _blood = Instantiate(bloodEffect, pos, rot); //피격 효과 프리팹을 충돌 지점에 생성
            Destroy(_blood, 1.0f); //생성된 피격 효과를 2초 후에 파괴

            
        }
```

<summary>BarrelCTRL스크립트에 SendMessage를 넣어서 에너미와 스왓에게 Die 라는 함수를 호출하도록 만들었음</summary>

```txt
   using Cinemachine;
using System.Collections;
using System.Collections.Generic;
using Unity.VisualScripting;
using UnityEngine;

public class BarrelCTRL : MonoBehaviour
{
    [SerializeField] private GameObject expEffect;
    [SerializeField]private Texture[] textures;// 텍스쳐 선언
    [SerializeField] private MeshRenderer renderer; // 텍스쳐를 바꿀 메쉬 랜더러 선언
    private Rigidbody rb; //리지드 바디 선언
    private int hitCount = 0; //총알에 맞은 횟수 파악
    public float expRadius = 20.0f; //폭파 반경
    private readonly string bulletTag = "BULLET"; //불릿 태그 선언
    public LayerMask layerMask; //레이어 마스크 선언
    public AudioSource source;
    [SerializeField] private AudioClip expclip;
    private MeshFilter filter;
    public Mesh[] meshes;
    private CinemachineImpulseSource impurseSource;
    void Start()
    {
        filter = GetComponent<MeshFilter>();
        
        renderer = GetComponent<MeshRenderer>(); // 프리팹 안에 있는 매쉬 랜더러 불러오기
        expclip = Resources.Load<AudioClip>("Sounds/missile_explosion");
        expEffect = Resources.Load<GameObject>("Effects/ExplosionMobile");
        textures = Resources.LoadAll<Texture>("Textures"); //리소스파일에 있는 텍스쳐 불러오기
        renderer.material.mainTexture = textures[Random.Range(0, textures.Length)]; 
        //매쉬랜더러의 머터리얼 안에 이미지가 들어있는 부분에 텍스쳐를 적용
        
    }
    private void OnCollisionEnter(Collision collision) //충돌 했을때
    {
        if (collision.gameObject.CompareTag(bulletTag)) //충돌한 게임 오브젝트가 불릿인제 확인
        {
            if (++hitCount == 3) //세번 맞았을 때
            {
                ExplosionBarrel(); //폭파 함수 출력
                if (impurseSource != null)//impurseSource가 존재한다면 유효성 검사라고 한다
                    impurseSource.GenerateImpulse(); //진동을 발생시키는 메서드 호출
            }
        }
        void ExplosionBarrel() //폭파 함수 정의
        {
            source.PlayOneShot(expclip,1.0f);
            var effect = Instantiate(expEffect, transform.position, Quaternion.identity);
            //폭파 이펙트를 var effect 로 넘김
            Destroy(effect, 2.0f); //2초후에 제거

            int idx = Random.Range(0, meshes.Length); //찌그러진 메쉬 적용
            filter.sharedMesh = meshes[idx];
            GetComponent<MeshFilter>().mesh = meshes[idx];

            Collider[] Cols = Physics.OverlapSphere(transform.position, expRadius);
            //15반경 안에 원으로 겹쳐진 layer들을 모두 Cols 라는 배열에 대입
            foreach (Collider col in Cols)
            {
                var _rb = col.attachedRigidbody;
                if (_rb == null) continue;
                _rb.mass = 1.0f; //무게를 잠시 1로 만듬
                //폭파력을 전달
                _rb.AddExplosionForce(1200f, transform.position, expRadius, 1000f);
                                     //폭발력 ,      위치 ,          반경 , 위로 솟구치는 힘
                col.gameObject.SendMessage("Die",SendMessageOptions.DontRequireReceiver);
            }
            Destroy(this.gameObject, 20.0f); //10초후에 삭제
        }
    }
}
```
